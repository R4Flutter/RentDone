## Authentication Patterns

### User Authentication Flow

```dart
// ✅ GOOD: Type-safe authentication with proper error handling

enum UserRole { owner, tenant, unknown }

class AuthController extends StateNotifier<AsyncValue<User?>> {
  AuthController(this._authService) : super(const AsyncValue.loading());

  final AuthService _authService;
  FirebaseAuth _firebaseAuth = FirebaseAuth.instance;

  // Stream-based auth state
  Stream<User?> authStateChanges() {
    return _firebaseAuth.authStateChanges();
  }

  // Login with error handling
  Future<void> login(String email, String password) async {
    try {
      // Validate inputs first
      _validateEmail(email);
      _validatePassword(password);

      // Attempt login
      final userCredential = await _firebaseAuth.signInWithEmailAndPassword(
        email: email.trim().toLowerCase(),
        password: password,
      );

      if (userCredential.user == null) {
        throw AuthException('User creation failed');
      }

      state = AsyncValue.data(userCredential.user);
    } on FirebaseAuthException catch (e) {
      // Handle Firebase-specific errors
      final message = _mapAuthError(e.code);
      state = AsyncValue.error(message, StackTrace.current);
    } on AuthException catch (e) {
      // Handle custom validation errors
      state = AsyncValue.error(e.message, StackTrace.current);
    } catch (e) {
      // Handle unexpected errors
      state = AsyncValue.error('Login failed', StackTrace.current);
    }
  }

  // Logout
  Future<void> logout() async {
    try {
      await _firebaseAuth.signOut();
      state = const AsyncValue.data(null);
    } catch (e) {
      state = AsyncValue.error('Logout failed', StackTrace.current);
    }
  }

  void _validateEmail(String email) {
    if (email.isEmpty) {
      throw AuthException('Email cannot be empty');
    }
    if (!email.contains('@')) {
      throw AuthException('Invalid email format');
    }
  }

  void _validatePassword(String password) {
    if (password.isEmpty) {
      throw AuthException('Password cannot be empty');
    }
    if (password.length < 6) {
      throw AuthException('Password must be at least 6 characters');
    }
  }

  String _mapAuthError(String code) {
    switch (code) {
      case 'user-not-found':
        return 'User not found. Please register first.';
      case 'wrong-password':
        return 'Incorrect password. Please try again.';
      case 'user-disabled':
        return 'This account has been disabled.';
      case 'too-many-requests':
        return 'Too many login attempts. Please try later.';
      default:
        return 'Login failed. Please try again.';
    }
  }
}

// Register as Riverpod provider
final authControllerProvider = StateNotifierProvider<AuthController, AsyncValue<User?>>((ref) {
  return AuthController(ref.watch(authServiceProvider));
});
```

### Google Sign-In Integration

```dart
// ✅ GOOD: Secure Google Sign-In with server-side verification

class GoogleAuthService {
  final GoogleSignIn _googleSignIn = GoogleSignIn(
    clientId: '...', // Use platform-specific client ID
  );
  
  final FirebaseAuth _firebaseAuth = FirebaseAuth.instance;

  Future<UserCredential?> signInWithGoogle() async {
    try {
      // 1. Trigger Google Sign-In flow
      final googleUser = await _googleSignIn.signIn();
      if (googleUser == null) {
        // User cancelled
        return null;
      }

      // 2. Get authentication tokens
      final googleAuth = await googleUser.authentication;

      // 3. Create Firebase credential from Google tokens
      final credential = GoogleAuthProvider.credential(
        accessToken: googleAuth.accessToken,
        idToken: googleAuth.idToken,
      );

      // 4. Sign in to Firebase
      final userCredential = await _firebaseAuth.signInWithCredential(credential);

      // 5. Verify user is registered in Firestore
      await _verifyUserInFirestore(userCredential.user!);

      return userCredential;
    } on FirebaseAuthException catch (e) {
      throw AuthException(_mapAuthError(e.code));
    } catch (e) {
      throw AuthException('Google sign-in failed');
    }
  }

  Future<void> _verifyUserInFirestore(User firebaseUser) async {
    final db = FirebaseFirestore.instance;
    final userDoc = await db.collection('users').doc(firebaseUser.uid).get();

    if (!userDoc.exists) {
      throw AuthException('User not found in system. Please register first.');
    }

    // Additional role verification can happen here
    final userData = userDoc.data() as Map<String, dynamic>;
    if (userData['role'] == null) {
      throw AuthException('Invalid user profile');
    }
  }

  String _mapAuthError(String code) {
    switch (code) {
      case 'account-exists-with-different-credential':
        return 'Email already registered with different provider';
      case 'invalid-credential':
        return 'Invalid credentials. Please try again.';
      default:
        return 'Google sign-in failed';
    }
  }
}
```

## Firestore Service Patterns

### Ownership Verification (Multi-Layer Security)

```dart
// ✅ GOOD: Ownership verification at service layer + rules

abstract class FirestoreService {
  FirebaseAuth get auth => FirebaseAuth.instance;
  FirebaseFirestore get db => FirebaseFirestore.instance;

  /// Get current authenticated user ID
  String getCurrentUserIdOrThrow() {
    final uid = auth.currentUser?.uid;
    if (uid == null) {
      throw UnauthorizedException('User must be authenticated');
    }
    return uid;
  }

  /// Verify user owns a resource
  Future<bool> verifyOwnership(String collectionPath, String docId, String ownerId) async {
    try {
      final doc = await db.collection(collectionPath).doc(docId).get();
      
      if (!doc.exists) {
        throw ResourceNotAccessibleException('Resource not found');
      }

      final data = doc.data() as Map<String, dynamic>;
      
      if (data['ownerId'] != ownerId) {
        throw InsufficientPermissionException('You do not own this resource');
      }

      return true;
    } catch (e) {
      rethrow;
    }
  }
}

// Example: Payment Service with ownership verification
class PaymentService extends FirestoreService {
  Future<void> updatePaymentStatus(
    String paymentId,
    PaymentStatus newStatus,
  ) async {
    final userId = getCurrentUserIdOrThrow();

    try {
      // Verify ownership before update
      await verifyOwnership('payments', paymentId, userId);

      // Update payment
      await db.collection('payments').doc(paymentId).update({
        'status': newStatus.toString(),
        'updatedAt': FieldValue.serverTimestamp(),
      });

      // Cache update locally
      await _cachePaymentUpdate(paymentId, newStatus);
    } on FirebaseException catch (e) {
      throw PaymentException.fromFirestore(e);
    }
  }

  Future<void> _cachePaymentUpdate(String paymentId, PaymentStatus status) async {
    final prefs = await SharedPreferences.getInstance();
    prefs.setString(
      'payment_status_$paymentId',
      status.toString(),
    );
  }
}
```

### Pagination for Large Datasets

```dart
// ✅ GOOD: Efficient pagination with caching

class TenantService extends FirestoreService {
  static const pageSize = 20;

  Future<(List<Tenant>, DocumentSnapshot?)> getTenantsPage({
    DocumentSnapshot? startAfter,
  }) async {
    final userId = getCurrentUserIdOrThrow();

    try {
      var query = db
          .collection('tenants')
          .where('ownerId', isEqualTo: userId)
          .orderBy('createdAt', descending: true)
          .limit(pageSize + 1);

      if (startAfter != null) {
        query = query.startAfterDocument(startAfter);
      }

      final snapshot = await query.get();
      
      // Convert to models
      final tenants = snapshot.docs
          .take(pageSize)  // Exclude the cursor doc
          .map((doc) => Tenant.fromFirestore(doc))
          .toList();

      // Return next cursor (if more results exist)
      final hasMore = snapshot.docs.length > pageSize;
      final nextCursor = hasMore ? snapshot.docs[pageSize - 1] : null;

      return (tenants, nextCursor);
    } on FirebaseException catch (e) {
      if (e.code == 'permission-denied') {
        throw InsufficientPermissionException('Cannot access tenants');
      }
      throw TenantException('Failed to fetch tenants');
    }
  }
}
```

### Error Handling with Retry Logic

```dart
// ✅ GOOD: Automatic retry with exponential backoff

class RetryableFirestoreService {
  static const maxRetries = 3;
  static const initialDelay = Duration(milliseconds: 100);

  Future<T> executeWithRetry<T>(
    Future<T> Function() operation, {
    String operationName = 'Operation',
  }) async {
    int attempt = 0;
    Duration delay = initialDelay;

    while (true) {
      try {
        return await operation();
      } on FirebaseException catch (e) {
        attempt++;

        // Don't retry on non-retriable errors
        if (!_isRetriable(e.code) || attempt >= maxRetries) {
          throw PaymentException.fromFirestore(e);
        }

        // Exponential backoff
        await Future.delayed(delay);
        delay *= 2;
      }
    }
  }

  bool _isRetriable(String code) {
    return ['unavailable', 'deadline-exceeded', 'resource-exhausted'].contains(code);
  }
}
```

## Caching & Offline Support

### Offline Cache Service

```dart
// ✅ GOOD: SharedPreferences-backed offline cache

class OfflineCacheService {
  late SharedPreferencesAsync prefs;

  Future<void> initialize() async {
    prefs = SharedPreferencesAsync();
  }

  /// Cache payment data
  Future<void> cachePayment(Payment payment) async {
    await prefs.setString(
      'payment_${payment.id}',
      jsonEncode(payment.toJson()),
    );
  }

  /// Get cached payment
  Future<Payment?> getCachedPayment(String paymentId) async {
    final cached = await prefs.getString('payment_$paymentId');
    if (cached == null) return null;

    try {
      return Payment.fromJson(jsonDecode(cached));
    } catch (e) {
      return null;
    }
  }

  /// Check cache freshness
  Future<bool> isCacheFresh(String paymentId, Duration maxAge) async {
    final timestamp = await prefs.getInt('payment_${paymentId}_timestamp');
    if (timestamp == null) return false;

    final age = DateTime.now().millisecondsSinceEpoch - timestamp;
    return age < maxAge.inMilliseconds;
  }

  /// Clear old cache
  Future<void> clearStaleCache(Duration maxAge) async {
    final keys = await prefs.getKeys();
    final now = DateTime.now().millisecondsSinceEpoch;

    for (final key in keys.where((k) => k.endsWith('_timestamp'))) {
      final timestamp = await prefs.getInt(key);
      if (timestamp != null && now - timestamp > maxAge.inMilliseconds) {
        await prefs.remove(key);
      }
    }
  }
}
```

### Offline-First Data Model

```dart
// ✅ GOOD: Models that support offline syncing

class Payment {
  final String id;
  final String tenantId;
  final double amount;
  final PaymentStatus status;
  
  /// Local-only fields for offline support
  final DateTime createdAt;
  final DateTime? syncedAt;  // null = not yet synced
  final String? syncError;   // null = no error

  bool get isPendingSync => syncedAt == null && syncError == null;
  bool get hasSyncError => syncError != null;

  Payment({
    required this.id,
    required this.tenantId,
    required this.amount,
    required this.status,
    required this.createdAt,
    this.syncedAt,
    this.syncError,
  });

  // Serialize for local storage
  Map<String, dynamic> toJson() => {
    'id': id,
    'tenantId': tenantId,
    'amount': amount,
    'status': status.string,
    'createdAt': createdAt.toIso8601String(),
    'syncedAt': syncedAt?.toIso8601String(),
    'syncError': syncError,
  };

  factory Payment.fromJson(Map<String, dynamic> json) => Payment(
    id: json['id'],
    tenantId: json['tenantId'],
    amount: json['amount'],
    status: PaymentStatus.values.firstWhere((s) => s.string == json['status']),
    createdAt: DateTime.parse(json['createdAt']),
    syncedAt: json['syncedAt'] != null ? DateTime.parse(json['syncedAt']) : null,
    syncError: json['syncError'],
  );

  // Serialize for Firestore
  Map<String, dynamic> toFirestore() => {
    'tenantId': tenantId,
    'amount': amount,
    'status': status.string,
    'createdAt': Timestamp.fromDate(createdAt),
    'updatedAt': FieldValue.serverTimestamp(),
  };
}
```

### Custom Exception Hierarchy

```dart
// ✅ GOOD: Type-safe, actionable error handling

abstract class AppException implements Exception {
  final String message;
  final String? code;

  AppException(this.message, {this.code});

  @override
  String toString() => message;
}

// Auth errors
class AuthException extends AppException {
  AuthException(String message, {String? code}) 
    : super(message, code: code);
}

class UnauthorizedException extends AuthException {
  UnauthorizedException(String message) 
    : super(message, code: 'unauthorized');
}

// Permission errors
class InsufficientPermissionException extends AppException {
  InsufficientPermissionException(String message) 
    : super(message, code: 'permission_denied');
}

// Payment errors
class PaymentException extends AppException {
  PaymentException(String message, {String? code})
    : super(message, code: code);

  factory PaymentException.fromFirestore(FirebaseException e) {
    switch (e.code) {
      case 'permission-denied':
        return PaymentException('Cannot perform operation', code: 'unauthorized');
      case 'not-found':
        return PaymentException('Payment not found', code: 'not_found');
      case 'already-exists':
        return PaymentException('Payment already exists', code: 'duplicate');
      default:
        return PaymentException('Payment operation failed', code: e.code);
    }
  }
}

// Error handling in service
Future<void> exampleOperation() async {
  try {
    // Perform operation
  } on UnauthorizedException {
    // Handle auth error
  } on InsufficientPermissionException {
    // Handle permission error
  } on PaymentException catch (e) {
    // Handle payment-specific error
    print('Payment error [${e.code}]: ${e.message}');
  } catch (e) {
    // Generic fallback
    throw PaymentException('Unexpected error: $e');
  }
}
```
